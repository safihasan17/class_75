create table brands
     (
     id int unsigned primary key auto_increment,
     Name varchar(100)
    );



create table categories
    (
     id int unsigned primary key auto_increment,
      Name varchar(100)
     );



create table products
       (
         id int unsigned primary key auto_increment,
        product_name varchar(100),
        brand_id int(11),
        category_id int(11),
         price float,
         is_active tinyint(4)
        );
